	.@target_map$ = getarg(0);
	.@mx = getarg(1);
	.@my = getarg(2);
	.@mob_id = getarg(3);
	.@mob_name$ = getarg(4);

	specialeffect2 154;
	message strcharinfo(0), .@mob_name$ + " has spawned in the center of the arena!";
	close;
}
