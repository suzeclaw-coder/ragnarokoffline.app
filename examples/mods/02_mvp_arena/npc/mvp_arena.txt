	mes "[MVP Boss Summoner]";
	mes "Hello, " + strcharinfo(0) + "!";
	mes "I can summon any MVP boss right beside you.";
	next;
	switch(select("~ Tier 1: Classic MVPs:~ Tier 2: Advanced MVPs:~ Tier 3: Endgame MVPs:~ Warp to Boss Field (prt_fild08):~ Clear All Spawned Monsters:~ Full Heal & Buffs:~ Cancel")) {
	case 1:
		mes "[Tier 1: Classic MVPs]";
		mes "Select a boss:";
		next;
		switch(select("Golden Thief Bug:Eddga:Moonlight Flower:Maya:Phreeoni:Orc Hero:Orc Lord:Drake:Mistress:Pharaoh:Osiris:Doppelganger:Super Porings (Angeling/Deviling/Ghostring/Mastering)")) {
			case 1: callsub SummonMob, 1086, "Golden Thief Bug";
			case 2: callsub SummonMob, 1115, "Eddga";
			case 3: callsub SummonMob, 1150, "Moonlight Flower";
			case 4: callsub SummonMob, 1147, "Maya";
			case 5: callsub SummonMob, 1159, "Phreeoni";
			case 6: callsub SummonMob, 1087, "Orc Hero";
			case 7: callsub SummonMob, 1190, "Orc Lord";
			case 8: callsub SummonMob, 1112, "Drake";
			case 9: callsub SummonMob, 1059, "Mistress";
			case 10: callsub SummonMob, 1157, "Pharaoh";
			case 11: callsub SummonMob, 1038, "Osiris";
			case 12: callsub SummonMob, 1046, "Doppelganger";
			case 13:
				.@map$ = strcharinfo(3);
				getmapxy(.@map$, .@px, .@py, BL_PC);
				monster .@map$, .@px+1, .@py, "Angeling", 1096, 1;
				monster .@map$, .@px-1, .@py, "Deviling", 1582, 1;
				monster .@map$, .@px, .@py+1, "Ghostring", 1120, 1;
				monster .@map$, .@px, .@py-1, "Mastering", 1090, 1;
				specialeffect2 154;
				message strcharinfo(0), "Super Porings have spawned around you!";
				close;
		}
		break;

	case 2:
		mes "[Tier 2: Advanced MVPs]";
		mes "Select a boss:";
		next;
		switch(select("Baphomet:Dark Lord:Lord of Death:Stormy Knight:Garm (Hatii):Evil Snake Lord:Incantation Samurai:Turtle General:Tao Gunka:RSX 0806:Amon Ra:Lady Tanee:Atroce:Ktullanux")) {
			case 1: callsub SummonMob, 1039, "Baphomet";
			case 2: callsub SummonMob, 1272, "Dark Lord";
			case 3: callsub SummonMob, 1373, "Lord of Death";
			case 4: callsub SummonMob, 1252, "Stormy Knight";
			case 5: callsub SummonMob, 1251, "Garm";
			case 6: callsub SummonMob, 1418, "Evil Snake Lord";
			case 7: callsub SummonMob, 1492, "Incantation Samurai";
			case 8: callsub SummonMob, 1312, "Turtle General";
			case 9: callsub SummonMob, 1583, "Tao Gunka";
			case 10: callsub SummonMob, 1623, "RSX 0806";
			case 11: callsub SummonMob, 1511, "Amon Ra";
			case 12: callsub SummonMob, 1688, "Lady Tanee";
			case 13: callsub SummonMob, 1785, "Atroce";
			case 14: callsub SummonMob, 1779, "Ktullanux";
		}
		break;

	case 3:
		mes "[Tier 3: Endgame MVPs]";
		mes "Select a boss:";
		next;
		switch(select("Valkyrie Randgris:Ifrit:Beelzebub:Fallen Bishop Hibram:Gloom Under Night:Kiel D-01:Nidhoggur's Shadow:Memory of Thanatos:Naght Sieger:Detale:Egnigem Cenia (Biolab):Wounded Morocc")) {
			case 1: callsub SummonMob, 1751, "Valkyrie Randgris";
			case 2: callsub SummonMob, 1832, "Ifrit";
			case 3: callsub SummonMob, 1874, "Beelzebub";
			case 4: callsub SummonMob, 1871, "Fallen Bishop Hibram";
			case 5: callsub SummonMob, 1768, "Gloom Under Night";
			case 6: callsub SummonMob, 1734, "Kiel D-01";
			case 7: callsub SummonMob, 2022, "Nidhoggur's Shadow";
			case 8: callsub SummonMob, 1708, "Memory of Thanatos";
			case 9: callsub SummonMob, 1957, "Naght Sieger";
			case 10: callsub SummonMob, 1719, "Detale";
			case 11: callsub SummonMob, 1658, "Egnigem Cenia";
			case 12: callsub SummonMob, 1917, "Wounded Morocc";
		}
		break;

	case 4:
		specialeffect2 12;
		close2;
		warp "prt_fild08", 170, 360;
		end;

	case 5:
		.@map$ = strcharinfo(3);
		killmonster .@map$, "All";
		specialeffect2 154;
		message strcharinfo(0), "All spawned monsters on " + .@map$ + " have been cleared!";
		close;

	case 6:
		percentheal 100, 100;
		sc_start SC_BLESSING, 3600000, 10;
		sc_start SC_INC_AGI, 3600000, 10;
		sc_start SC_KYRIE, 3600000, 10;
		sc_start SC_MAGNIFICAT, 3600000, 5;
		repairall;
		specialeffect2 20;
		message strcharinfo(0), "Full HP/SP and buffs applied!";
		close;

	case 7:
		close;
	}
	end;

SummonMob:
	.@mob_id = getarg(0);
	.@mob_name$ = getarg(1);
	.@map$ = strcharinfo(3);
	getmapxy(.@map$, .@px, .@py, BL_PC);
	monster .@map$, .@px, .@py, .@mob_name$, .@mob_id, 1;
	specialeffect2 154;
	message strcharinfo(0), .@mob_name$ + " has spawned right beside you!";
	close;
}
