prontera,147,195,4	script	Stat & Skill Resetter#prt	4_F_SISTER,{
	mes "[Stat & Skill Resetter]";
	mes "Hello " + strcharinfo(0) + "!";
	mes "I can reset your stats or skill points completely free of charge.";
	next;
	switch(select("~ Reset Status Points:~ Reset Skill Points:~ Reset Both:~ Cancel")) {
	case 1:
		resetstatus;
		specialeffect2 154;
		message strcharinfo(0), "Your stat points have been refunded!";
		break;
	case 2:
		resetskill;
		specialeffect2 154;
		message strcharinfo(0), "Your skill points have been refunded!";
		break;
	case 3:
		resetstatus;
		resetskill;
		specialeffect2 154;
		message strcharinfo(0), "Both stats and skills have been reset!";
		break;
	case 4:
		close;
	}
	close;
}
